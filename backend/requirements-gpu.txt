# GPU support for face detection (InsightFace) and quality scoring
# (MagFace) via onnxruntime's CUDA execution provider.
#
# Requires an NVIDIA GPU + driver supporting CUDA 12.x (check with
# `nvidia-smi` - the "CUDA Version" it reports just needs to be >= 12.x,
# the actual CUDA/cuDNN runtime comes from the pip packages below, no
# system CUDA toolkit install needed).
#
# Install *after* requirements.txt, then remove the CPU onnxruntime
# build so its files don't shadow the GPU one (both install into the
# same "onnxruntime" package on disk):
#
#   pip install -r requirements.txt
#   pip uninstall -y onnxruntime
#   pip install -r requirements-gpu.txt
#
# Then set FACE_USE_GPU=true / FACE_QUALITY_USE_GPU=true (see
# .env.example). No other setup needed - app/core/gpu_env.py locates
# these packages' bundled .so files and wires them up automatically at
# startup.
onnxruntime-gpu==1.24.1
nvidia-cublas-cu12==12.9.2.10
nvidia-cuda-nvrtc-cu12==12.9.86
nvidia-cuda-runtime-cu12==12.9.79
nvidia-cudnn-cu12==9.24.0.43
nvidia-cufft-cu12==11.4.1.4
nvidia-curand-cu12==10.3.10.19
